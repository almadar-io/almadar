# التدويل (i18n)

يدعم المدار كتابة المخططات بلغتك الأم. تكتشف أداة سطر الأوامر (CLI) اللغة تلقائيًا بناءً على الكلمات المفتاحية وتقوم بترجمتها إلى التمثيل الداخلي القياسي قبل التجميع.

## الاستخدام

ما عليك سوى كتابة المخطط الخاص بك باستخدام الكلمات المفتاحية والعوامل المترجمة للغتك. يظل الامتداد كما هو `.orb`.

### مثال

```json
{
  "اسم": "مرحبا_بالعالم",
  "مدارات": [{
    "اسم": "محيا",
    "كيان": {
      "اسم": "تحية",
      "حقول": [
        { "اسم": "رسالة", "نوع": "نص" },
        { "اسم": "عدد", "نوع": "رقم", "افتراضي": 0 }
      ]
    },
    "سمات": [{
      "اسم": "قابل_للنقر",
      "آلة_حالة": {
        "حالات": [
          { "اسم": "خامل", "أولي": true },
          { "اسم": "مرحب_به" }
        ],
        "انتقالات": [{
          "من": "خامل",
          "حدث": "نقر",
          "إلى": "مرحب_به",
          "تأثيرات": [
            ["تعيين", "@كيان.رسالة", "مرحبا، أيها العالم!"],
            ["تعيين", "@كيان.عدد", ["+", "@كيان.عدد", 1]],
            ["رسم_واجهة", "رئيسي", {
              "نوع": "stats-card",
              "العنوان": "@كيان.رسالة",
              "القيمة": "@كيان.عدد"
            }]
          ]
        }]
      }
    }]
  }]
}
```
